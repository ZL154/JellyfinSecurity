> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add internationalization infrastructure (translation endpoint, embedded resource files, JS helper) and seed translations for 8 languages: de, en, es, fr, it, ja, pt, zh. Wire `tr(key, fallback)` into the highest-impact admin UI strings as a proof-of-integration; defer deep coverage of every English string to follow-up v2.5.x patches.

**Architecture:** Mirrors the AchievementBadges precedent verbatim. Embedded resource JSON files per language → `GET /TwoFactorAuth/translations/{lang}` controller endpoint (sanitized + cache-disabled + English fallback) → `tr(key, fallback)` JS helper that consults a loaded bundle and falls back to the inline fallback if the key is missing. Per-user `Language` preference field + admin `DefaultLanguage` config + small preference endpoint (read/write).

**Tech Stack:** C# / .NET 9, JSON files as `<EmbeddedResource>`, controller endpoint, vanilla JS helper, xUnit tests for the endpoint sanitization and fallback paths.

**Scope explicitly out:** Re-keying every English string in admin.html (~1700 lines of JS); challenge.html and setup.html `tr()` wiring; inject.js translations. These remain English in v2.5.0 and ride follow-up patch releases — the goal here is infrastructure + a credible critical-strings seed.

```csharp
namespace Jellyfin.Plugin.TwoFactorAuth.Helpers;

/// <summary>Reads bytes / text out of the assembly's embedded resources.
/// Centralizes the GetManifestResourceStream + StreamReader dance.</summary>
public static class ResourceReader
{
    public static string? ReadEmbeddedText(string logicalName)
    {
        var asm = typeof(ResourceReader).Assembly;
        using var stream = asm.GetManifestResourceStream(logicalName);
```

```csharp
[AllowAnonymous]
public ActionResult GetTranslations([FromRoute] string lang)
{
    // v2.5.0: no auth required so challenge.html and setup.html can load
    // translations before the user has signed in. Sanitize lang to prevent
    // path traversal - only letters and dashes allowed.
    Response.Headers["Cache-Control"] = "no-cache, must-revalidate";
    var clean = new string((lang ?? "en").ToLowerInvariant()
        .Where(c => (c >= 'a' && c <= 'z') || c == '-').ToArray());
    if (string.IsNullOrEmpty(clean)) clean = "en";

    var content = Helpers.ResourceReader.ReadEmbeddedText(
        $"Jellyfin.Plugin.TwoFactorAuth.Pages.translations.{clean}.json");
    if (content is null && clean != "en")
    {
        content = Helpers.ResourceReader.ReadEmbeddedText(
            "Jellyfin.Plugin.TwoFactorAuth.Pages.translations.en.json");
    }
    if (content is null) return NotFound();
    return Content(content, "application/json");
}
```

```csharp
[HttpPut("users/{userId:guid}/preferences")]
[Authorize]
public async Task<IActionResult> UpdateUserPreferences([FromRoute] Guid userId, [FromBody] UpdatePreferencesRequest request)
{
    if (!IsAuthorizedForUser(userId)) return Forbid();
    if (request is null) return BadRequest(new { message = "body required" });

    if (!string.IsNullOrEmpty(request.Language))
    {
        var allowed = new[] { "en", "de", "es", "fr", "it", "ja", "pt", "zh" };
        if (Array.IndexOf(allowed, request.Language) < 0)
            return BadRequest(new { message = "unsupported language" });
    }

    await _store.MutateAsync(userId, d =>
    {
        d.Language = string.IsNullOrEmpty(request.Language) ? null : request.Language;
    }).ConfigureAwait(false);

    return Ok(new { ok = true });
}
```

```javascript
// v2.5.0 i18n: minimal tr(key, fallback) helper.
var _tfaTranslations = null;
function tr(key, fallback) {
    if (_tfaTranslations && Object.prototype.hasOwnProperty.call(_tfaTranslations, key)) {
        return _tfaTranslations[key];
    }
    return fallback != null ? fallback : key;
}
async function loadTfaTranslations() {
    try {
        var uid = ApiClient.getCurrentUserId();
        var prefP = uid
            ? ApiClient.ajax({ type: 'GET', url: ApiClient.getUrl('TwoFactorAuth/users/' + uid + '/preferences'), dataType: 'json' }).catch(() => null)
            : Promise.resolve(null);
        var pref = await prefP;
        var lang = (pref && pref.effectiveLanguage) || 'en';
        // Sanitize on the client too so a tampered cookie / URL doesn't load garbage.
        lang = String(lang).toLowerCase().replace(/[^a-z-]/g, '') || 'en';
        var resp = await fetch(ApiClient.getUrl('TwoFactorAuth/translations/' + encodeURIComponent(lang)), { credentials: 'include' });
        if (!resp.ok) return;
        _tfaTranslations = await resp.json();
        applyTranslations();
    } catch (e) { console.warn('translation load failed', e); }
}
function applyTranslations() {
```
